function instalarSistemaPesquisaSatisfacao() {
  const planilha = SpreadsheetApp.getActiveSpreadsheet();
  atualizarPesquisaSatisfacao(planilha);
  instalarGatilhoPesquisaSatisfacao_(planilha);
}

function atualizarPesquisaSatisfacao(planilhaInformada, opcoes) {
  if (planilhaInformada && typeof planilhaInformada.getSheetByName !== 'function') {
    opcoes = planilhaInformada;
    planilhaInformada = null;
  }
  opcoes = opcoes || {};
  if (!opcoes.baseJaSincronizada && typeof sincronizarBaseConsolidada1e3 === 'function') {
    sincronizarBaseConsolidada1e3({origem: 'pesquisa de satisfação'});
  }

  const bloqueio = LockService.getDocumentLock();
  bloqueio.waitLock(30000);

  try {
    const planilha = planilhaInformada || SpreadsheetApp.getActiveSpreadsheet();
    if (!planilha) {
      throw new Error('Não foi possível identificar a planilha ativa.');
    }

    const registros = extrairRegistrosPesquisaUnificados_(planilha);
    const indicadores = calcularIndicadoresPesquisa_(registros);
    const fusoHorario = planilha.getSpreadsheetTimeZone();
    const abaBi = obterOuCriarAbaAnalitica_(planilha, CONFIG_PESQUISA_SATISFACAO.ABA_BI);
    const periodoNaAba = obterPeriodoGravadoNaAbaPesquisa_(abaBi);
    const periodoSolicitado = opcoes.periodoSelecionado
      || (opcoes.silencioso && opcoes.baseJaSincronizada ? '' : periodoNaAba);
    const contextoMensal = construirContextoMensalPesquisa_(
      registros,
      fusoHorario,
      periodoSolicitado
    );

    if (!opcoes.apenasAnaliticas) {
      const aba = obterOuCriarAbaPesquisa_(planilha);
      prepararAbaPesquisa_(aba);
      escreverPainelPesquisa_(aba, indicadores, registros, fusoHorario);
      escreverBasePesquisa_(aba, registros);
      formatarAbaPesquisa_(aba, registros.length);
    }

    const abaParceiros = obterOuCriarAbaAnalitica_(planilha, CONFIG_PESQUISA_SATISFACAO.ABA_PARCEIROS);
    prepararAbaAnalitica_(abaParceiros, 15);
    escreverProgramaParceiros_(abaParceiros, registros, indicadores, fusoHorario, contextoMensal);

    prepararAbaAnalitica_(abaBi, 16);
    escreverBISatisfacao_(abaBi, registros, indicadores, contextoMensal, fusoHorario);
    PropertiesService.getDocumentProperties().setProperty(
      CONFIG_PESQUISA_SATISFACAO.PROPRIEDADE_PERIODO,
      contextoMensal.atual.chave || ''
    );
  } finally {
    bloqueio.releaseLock();
  }
}

function aoEnviarFormularioPesquisaSatisfacao_(evento) {
  const planilha = evento && evento.source
    ? evento.source
    : SpreadsheetApp.getActiveSpreadsheet();
  if (typeof sincronizarBaseConsolidada1e3 === 'function') {
    sincronizarBaseConsolidada1e3(evento || {origem: 'gatilho de satisfação'});
    atualizarPesquisaSatisfacao(planilha, {baseJaSincronizada: true, silencioso: true});
    return;
  }
  atualizarPesquisaSatisfacao(planilha, {silencioso: true});
}

function aoEditarPeriodoBISatisfacao_(evento) {
  if (!evento || !evento.range || !evento.source) return;
  const aba = evento.range.getSheet();
  if (aba.getName() !== CONFIG_PESQUISA_SATISFACAO.ABA_BI) return;
  if (evento.range.getA1Notation() !== CONFIG_PESQUISA_SATISFACAO.CELULA_PERIODO_BI) return;
  atualizarPesquisaSatisfacao(evento.source, {
    baseJaSincronizada: true,
    apenasAnaliticas: true,
    periodoSelecionado: evento.value || evento.range.getDisplayValue(),
    silencioso: true
  });
}

function atualizarPainelMensalSatisfacao() {
  atualizarPesquisaSatisfacao(SpreadsheetApp.getActiveSpreadsheet(), {
    baseJaSincronizada: true,
    apenasAnaliticas: true
  });
}

function instalarGatilhoPesquisaSatisfacao_(planilha) {
  ScriptApp.getProjectTriggers().forEach(function(gatilho) {
    const funcao = gatilho.getHandlerFunction();
    if (funcao === CONFIG_PESQUISA_SATISFACAO.FUNCAO_GATILHO ||
        funcao === CONFIG_PESQUISA_SATISFACAO.FUNCAO_GATILHO_EDICAO) {
      ScriptApp.deleteTrigger(gatilho);
    }
  });

  if (typeof aoEnviarFormularioIntegracaoProcedimentos !== 'function') {
    ScriptApp.newTrigger(CONFIG_PESQUISA_SATISFACAO.FUNCAO_GATILHO)
      .forSpreadsheet(planilha)
      .onFormSubmit()
      .create();
  }
  ScriptApp.newTrigger(CONFIG_PESQUISA_SATISFACAO.FUNCAO_GATILHO_EDICAO)
    .forSpreadsheet(planilha)
    .onEdit()
    .create();
}

function localizarAbaOrigemPesquisa_(planilha) {
  const consolidada = planilha.getSheetByName(CONFIG_PESQUISA_SATISFACAO.ABA_CONSOLIDADA);
  if (!consolidada) {
    throw new Error('A Base Consolidada não foi encontrada. Execute primeiro a integração.');
  }
  return consolidada;
}

function extrairRegistrosPesquisaUnificados_(planilha) {
  const consolidada = localizarAbaOrigemPesquisa_(planilha);
  const dados = consolidada.getDataRange().getValues();
  if (dados.length < 2) return [];
  const cabecalhos = dados[0].map(function(valor) {
    return String(valor || '').trim();
  });
  const mapa = criarMapaCabecalhosPesquisa_(cabecalhos);
  const registros = [];
  const chavesVistas = {};
  extrairRegistrosPesquisa_(dados, mapa).forEach(function(registro) {
    const chave = construirChaveRegistroPesquisa_(registro);
    if (chavesVistas[chave]) return;
    chavesVistas[chave] = true;
    registros.push(registro);
  });

  registros.sort(function(a, b) {
    const dataA = a.data instanceof Date ? a.data.getTime() : 0;
    const dataB = b.data instanceof Date ? b.data.getTime() : 0;
    const envioA = a.registroEnvio instanceof Date ? a.registroEnvio.getTime() : 0;
    const envioB = b.registroEnvio instanceof Date ? b.registroEnvio.getTime() : 0;
    return dataB - dataA || envioB - envioA || b.id.localeCompare(a.id);
  });
  return registros;
}

function criarMapaCabecalhosPesquisa_(cabecalhos) {
  const mapa = {};
  cabecalhos.forEach(function(cabecalho, indice) {
    const chave = normalizarPesquisa_(cabecalho);
    if (chave && mapa[chave] === undefined) {
      mapa[chave] = indice;
    }
  });
  return mapa;
}

  const indiceOrigem = mapa[normalizarPesquisa_(CONFIG_PESQUISA_SATISFACAO.CABECALHO_ORIGEM_REGISTRO)];
  const indiceIdMigracao = mapa[normalizarPesquisa_(CONFIG_PESQUISA_SATISFACAO.CABECALHO_ID_MIGRACAO)];

    const linha = dados[indiceLinha];
    const procedimento = buscarValorPesquisa_(linha, mapa, cabecalhoProcedimento);

        !temConteudoPesquisaSatisfacao_(linha, mapa)) {
      continue;

      'Perguntas finais direcionadas ao momento, indique se:'

      'Data de realização do procedimento'
    ]));
    if (!dataRealizacao) {
      Logger.log(
        'Pesquisa ignorada por ausência de "Data de realização do procedimento". Linha: ' +
        (indiceLinha + 1)
      );
      continue;
    }
    const idMigracao = indiceIdMigracao === undefined
      ? ''
      : textoPesquisa_(linha[indiceIdMigracao]);

    const registro = {

      origemRegistro: indiceOrigem === undefined ? '' : textoPesquisa_(linha[indiceOrigem]),
      bairro: textoPesquisa_(buscarValorPesquisa_(linha, mapa, ['Bairro de realização do procedimento'])),
      endereco: textoPesquisa_(buscarValorPesquisa_(linha, mapa, ['Endereço completo'])),
      referencia: textoPesquisa_(buscarValorPesquisa_(linha, mapa, ['Ponto de referência'])),
      nome: textoPesquisa_(buscarValorPesquisa_(linha, mapa, ['Nome - Pesquisa de Satisfação'])),
      telefone: formatarTelefonePesquisa_(buscarValorPesquisa_(linha, mapa, ['Telefone - Pesquisa de Satisfação'])),
      momento: textoPesquisa_(momento),
      notaSaneamento: numeroPesquisa_(buscarValorPesquisa_(linha, mapa, [
        'De 0 a 10, quanto está satisfeito com os serviços de saneamento de forma geral?'
      ])),
      aspectos: textoPesquisa_(buscarValorPesquisa_(linha, mapa, [
        'De forma geral, quais aspectos observa sobre os serviços de saneamento?'
      ])),
      mudancaQualidade: textoPesquisa_(buscarValorPesquisa_(linha, mapa, [
        'Tem notado mudanças na qualidade de vida em relação ao saneamento de forma geral?'
      ])),
      notaMelhoria: numeroPesquisa_(buscarValorPesquisa_(linha, mapa, [
        'Se sim: De 0 a 10, quanto percebeu melhorias na qualidade de vida relacionadas ao saneamento?'
      ])),
      notaEtapa: numeroPesquisa_(notaEtapa),
      relacionamento: numeroPesquisa_(buscarValorPesquisa_(linha, mapa, [
        'De 0 a 10, como avalia o relacionamento entre a equipe de obras e a população?'
      ])),
      execucao: numeroPesquisa_(buscarValorPesquisa_(linha, mapa, [
        'De 0 a 10, o quanto está satisfeito com a execução da obra?'
      ])),
      expectativas: textoPesquisa_(buscarValorPesquisa_(linha, mapa, ['As obras atenderam as expectativas?'])),
      resultado: numeroPesquisa_(buscarValorPesquisa_(linha, mapa, [
        'De 0 a 10, o quanto está satisfeito com o resultado da obra?'
      ])),
      atendimentoSabesp: numeroPesquisa_(buscarValorPesquisa_(linha, mapa, [
        'De 0 a 10, como avalia o atendimento da Sabesp?'
      ])),
      clareza: textoPesquisa_(buscarValorPesquisa_(linha, mapa, ['As informações ficaram claras?'])),
      notaAtendimento: numeroPesquisa_(buscarValorPesquisa_(linha, mapa, [
        'De 0 a 10, qual nota o munícipe atribui ao atendimento recebido?'
      ])),
      interesseContato: textoPesquisa_(buscarValorPesquisa_(linha, mapa, [
        'Tem interesse em contato posterior sobre as Obras?'
      ])),
      interesseGrupo: textoPesquisa_(buscarValorPesquisa_(linha, mapa, [
        'Gostaria de fazer parte de um grupo de comunicação sobre as obras e ações do Consórcio?'
      ])),
      sugestaoSaneamento: textoPesquisa_(buscarValorPesquisa_(linha, mapa, [

      impactoDiaADia: textoPesquisa_(buscarValorPesquisa_(linha, mapa, [
        'Deseja relatar como as obras vem afetando seu dia a dia?'
      ])),
      sugestaoSabesp: textoPesquisa_(buscarValorPesquisa_(linha, mapa, [
        'Deseja registrar alguma sugestão ou comentário sobre os serviços da Sabesp?'
      ])),
      comentarioFinal: textoPesquisa_(buscarValorPesquisa_(linha, mapa, [

  }

  return registros;
}

function construirChaveRegistroPesquisa_(registro) {
  if (textoPesquisa_(registro.id)) {
    return 'id|' + normalizarPesquisa_(registro.id);
  }
  return [
    registro.registroEnvio instanceof Date ? registro.registroEnvio.getTime() : '',
    registro.data instanceof Date ? registro.data.getTime() : '',
    normalizarPesquisa_(registro.nome),
    normalizarPesquisa_(registro.telefone),
    normalizarPesquisa_(registro.endereco),
    registro.notaSaneamento === null ? '' : registro.notaSaneamento,
    normalizarPesquisa_(registro.aspectos),
    normalizarPesquisa_(registro.comentarios)
  ].join('|');
}

function temConteudoPesquisaSatisfacao_(linha, mapa) {
  const camposAncora = [
    'Nome - Pesquisa de Satisfação',
    'De 0 a 10, quanto está satisfeito com os serviços de saneamento de forma geral?',
    'De forma geral, quais aspectos observa sobre os serviços de saneamento?',
    'Tem notado mudanças na qualidade de vida em relação ao saneamento de forma geral?'
  ];
  return camposAncora.some(function(cabecalho) {
    const valor = buscarValorPesquisa_(linha, mapa, [cabecalho]);
    return valor !== null && valor !== undefined && String(valor).trim() !== '';
  });
}

function obterNotaEtapaPesquisa_(linha, mapa, momento) {
  const momentoNormalizado = normalizarPesquisa_(momento);
  if (momentoNormalizado.indexOf('pre obra') !== -1) {
    return buscarValorPesquisa_(linha, mapa, ['De 0 a 10, o quanto está satisfeito com a chegada da obra?']);
  }
  if (momentoNormalizado.indexOf('durante') !== -1) {
    return buscarValorPesquisa_(linha, mapa, ['De 0 a 10, o quanto está satisfeito com a execução da obra?']);
  }
  if (momentoNormalizado.indexOf('pos obra') !== -1) {
    return buscarValorPesquisa_(linha, mapa, ['De 0 a 10, o quanto está satisfeito com o resultado da obra?']);
  }

  const alternativas = [
    'De 0 a 10, o quanto está satisfeito com a chegada da obra?',
    'De 0 a 10, o quanto está satisfeito com a execução da obra?',
    'De 0 a 10, o quanto está satisfeito com o resultado da obra?'
  ];

  for (let i = 0; i < alternativas.length; i++) {
    const valor = buscarValorPesquisa_(linha, mapa, [alternativas[i]]);
    if (numeroPesquisa_(valor) !== null) {
      return valor;
    }
  }
  return null;
}

function consolidarComentariosPesquisa_(linha, mapa) {
  const campos = [
    ['Melhoria dos serviços', 'Quer adicionar algum comentário ou sugestão para melhoria dos serviços?'],
    ['Redução de impactos', 'Tem algum comentário ou sugestão que ajude a causar menos impactos com chegada da obra?'],
    ['Impacto no dia a dia', 'Deseja relatar como as obras vem afetando seu dia a dia?'],
    ['Serviços da Sabesp', 'Deseja registrar alguma sugestão ou comentário sobre os serviços da Sabesp?'],
    ['Comentário final', 'Algum último comentário ou sugestão?'],
    ['Observação do registro', 'Observação final do procedimento']
  ];
  const partes = [];

  campos.forEach(function(campo) {
    const valor = buscarValorPesquisa_(linha, mapa, [campo[1]]);
    if (possuiConteudoPesquisa_(valor)) {
      partes.push(campo[0] + ': ' + String(valor).trim());
    }
  });

  return partes.join('\n');
}

function extrairTemasPesquisa_(registro) {
  const texto = normalizarPesquisa_([
    registro.aspectos,
    registro.sugestaoSaneamento,
    registro.impactoDiaADia,
    registro.sugestaoSabesp,
    registro.comentarioFinal,
    registro.observacaoFinal
  ].join(' '));

  const regras = obterRegrasTemasPesquisa_();

  return regras.filter(function(regra) {
    return regra[1].some(function(termo) {
      return contemTermoPesquisa_(texto, termo);
    });
  }).map(function(regra) {
    return regra[0];
  });
}

function obterRegrasTemasPesquisa_() {
  return [
    ['Funcionamento positivo', ['bom funcionamento', 'boas condicoes', 'nao percebe problemas', 'nunca teve problemas']],
    ['Abastecimento de água', ['agua', 'racionamento', 'religacao', 'abastecimento']],
    ['Limpeza e rios', ['limpeza', 'rio', 'rios']],
    ['Vetores', ['rato', 'ratos', 'escorpiao', 'escorpioes', 'vetor', 'vetores']],
    ['Pavimentação', ['pavimentacao', 'pavimento']],
    ['Atendimento e burocracia', ['burocracia', 'demora no atendimento']],
    ['Espaço público', ['praca', 'espaco publico']]
  ];
}

function contemTermoPesquisa_(texto, termo) {
  return (' ' + texto + ' ').indexOf(' ' + normalizarPesquisa_(termo) + ' ') !== -1;
}

function possuiConteudoPesquisa_(valor) {
  const normalizado = normalizarPesquisa_(valor);
  if (!normalizado) {
    return false;
  }

  const semConteudo = [
    'n', 'nao', 'ni', 'nao informado', 'sem comentario', 'sem comentarios',
    'sem observacao', 'sem observacoes', 'nao possui', 'nada a relatar'
  ];
  return semConteudo.indexOf(normalizado) === -1;
}

function calcularIndicadoresPesquisa_(registros) {
  const notasSaneamento = numerosValidosPesquisa_(registros.map(function(item) {
    return item.notaSaneamento;
  }));
  const notasMelhoria = numerosValidosPesquisa_(registros.map(function(item) {
    return item.notaMelhoria;
  }));
  const notasAtendimento = numerosValidosPesquisa_(registros.map(function(item) {
    return item.notaAtendimento;
  }));
  const respostasClareza = registros.filter(function(item) {
    return normalizarPesquisa_(item.clareza) !== '';
  });
  const respostasClaras = respostasClareza.filter(function(item) {
    return normalizarPesquisa_(item.clareza) === 'sim';
  });
  const notasAltas = notasAtendimento.filter(function(nota) { return nota >= 9; }).length;
  const respostasMudanca = registros.filter(function(item) {
    return normalizarPesquisa_(item.mudancaQualidade) !== '';
  });
  const mudancaSim = respostasMudanca.filter(function(item) {
    return normalizarPesquisa_(item.mudancaQualidade) === 'sim';
  });

  return {
    total: registros.length,
    mediaSaneamento: mediaPesquisa_(notasSaneamento),
    mediaMelhoria: mediaPesquisa_(notasMelhoria),
    mediaAtendimento: mediaPesquisa_(notasAtendimento),
    notasAltas: notasAtendimento.length
      ? Math.round((notasAltas / notasAtendimento.length) * 100)
      : null,
    quantidadeNotasAltas: notasAltas,
    clareza: respostasClareza.length
      ? Math.round((respostasClaras.length / respostasClareza.length) * 100)
      : null,
    mudancaPercebida: respostasMudanca.length
      ? Math.round((mudancaSim.length / respostasMudanca.length) * 100)
      : null,
    respostasSaneamento: notasSaneamento.length,
    respostasMelhoria: notasMelhoria.length,
    respostasAtendimento: notasAtendimento.length,
    respostasClareza: respostasClareza.length,
    respostasMudanca: respostasMudanca.length
  };
}

function calcularTermometroPesquisa_(registros) {
  const notasSaneamento = numerosValidosPesquisa_(registros.map(function(item) { return item.notaSaneamento; }));
  const notasMelhoria = numerosValidosPesquisa_(registros.map(function(item) { return item.notaMelhoria; }));
  const notasAtendimento = numerosValidosPesquisa_(registros.map(function(item) { return item.notaAtendimento; }));
  const respostasMudanca = registros.filter(function(item) {
    return normalizarPesquisa_(item.mudancaQualidade) !== '';
  });
  const respostasClareza = registros.filter(function(item) {
    return normalizarPesquisa_(item.clareza) !== '';
  });
  const respostasAspectos = registros.filter(function(item) {
    return normalizarPesquisa_(item.aspectos) !== '';
  });
  const respostasClassificacao = registros.filter(function(item) {
    return normalizarPesquisa_(item.classificacao) !== '';
  });
  const aspectos = contarAspectosQuestionarioPesquisa_(registros);
  const problemasAbertos = contarProblemasAbertosPesquisa_(registros);
  const classificacoes = contarClassificacoesPesquisa_(registros);

  const saneamentoAte6 = notasSaneamento.filter(function(nota) { return nota <= 6; }).length;
  const saneamento9ou10 = notasSaneamento.filter(function(nota) { return nota >= 9; }).length;
  const melhoriaAte6 = notasMelhoria.filter(function(nota) { return nota <= 6; }).length;
  const semMudanca = respostasMudanca.filter(function(item) {
    return normalizarPesquisa_(item.mudancaQualidade) === 'nao';
  }).length;
  const informacaoNaoPlena = respostasClareza.filter(function(item) {
    return normalizarPesquisa_(item.clareza) !== 'sim';
  }).length;
  const inconsistenciasQualidade = registros.filter(function(item) {
    return normalizarPesquisa_(item.mudancaQualidade) === 'nao'
      && typeof item.notaMelhoria === 'number'
      && Number.isFinite(item.notaMelhoria);
  }).length;
  const comAspectoNegativo = respostasAspectos.filter(function(item) {
    return possuiAspectoNegativoPesquisa_(item.aspectos);
  }).length;
  const comAspectoPositivo = respostasAspectos.filter(function(item) {
    return possuiAspectoPositivoPesquisa_(item.aspectos);
  }).length;

  return {
    saneamentoAte6: saneamentoAte6,
    baseSaneamento: notasSaneamento.length,
    saneamento9ou10: saneamento9ou10,
    melhoriaAte6: melhoriaAte6,
    baseMelhoria: notasMelhoria.length,
    semMudanca: semMudanca,
    baseMudanca: respostasMudanca.length,
    informacaoNaoPlena: informacaoNaoPlena,
    baseClareza: respostasClareza.length,
    aspectoNegativo: comAspectoNegativo,
    aspectoPositivo: comAspectoPositivo,
    baseAspectos: respostasAspectos.length,
    reclamacoes: classificacoes.Reclamação || 0,
    baseClassificacoes: respostasClassificacao.length,
    inconsistenciasQualidade: inconsistenciasQualidade,
    baseInconsistencia: registros.length,
    notasAtendimento: notasAtendimento.length,
    aspectos: aspectos,
    problemasAbertos: problemasAbertos,
    classificacoes: classificacoes
  };
}

function contarAspectosQuestionarioPesquisa_(registros) {
  const regras = obterRegrasAspectosQuestionarioPesquisa_();
  const resultado = {};
  regras.forEach(function(regra) { resultado[regra.chave] = 0; });
  registros.forEach(function(registro) {
    const texto = normalizarPesquisa_(registro.aspectos);
    regras.forEach(function(regra) {
      if (regra.termos.some(function(termo) { return contemTermoPesquisa_(texto, termo); })) {
        resultado[regra.chave]++;
      }
    });
  });
  return resultado;
}

  const texto = normalizarPesquisa_(aspectos);

      && regra.termos.some(function(termo) { return contemTermoPesquisa_(texto, termo); });
  });
}

function possuiAspectoPositivoPesquisa_(aspectos) {
  const texto = normalizarPesquisa_(aspectos);
  return obterRegrasAspectosQuestionarioPesquisa_().some(function(regra) {
    return regra.natureza === 'Positivo'
      && regra.termos.some(function(termo) { return contemTermoPesquisa_(texto, termo); });
  });
}

function contarProblemasAbertosPesquisa_(registros) {
  const resultado = {};

  registros.forEach(function(registro) {

      registro.aspectos, registro.sugestaoSaneamento, registro.impactoDiaADia,

        resultado[regra[0]]++;
      }
    });
  });
  return resultado;
}

function formatarVariacaoQuantidadePesquisa_(atual, anterior, rotuloAnterior) {
  const diferenca = atual - anterior;
  if (diferenca === 0) return 'mesmo volume de ' + rotuloAnterior;
  return Math.abs(diferenca) + (diferenca > 0 ? ' a mais' : ' a menos') + ' que ' + rotuloAnterior;
}

function formatarVariacaoMediaPesquisa_(atual, anterior, rotuloAnterior) {
  if (atual === null || anterior === null || !Number.isFinite(atual) || !Number.isFinite(anterior)) {
    return 'sem base comparável em ' + rotuloAnterior;
  }
  const diferenca = Math.round((atual - anterior) * 10) / 10;
  if (diferenca === 0) return 'estável em relação a ' + rotuloAnterior;
  return (diferenca > 0 ? 'aumento de ' : 'redução de ') + Math.abs(diferenca).toFixed(1).replace('.', ',')
    + (Math.abs(diferenca) === 1 ? ' ponto' : ' pontos') + ' em relação a ' + rotuloAnterior;
}

function formatarQuantidadePesquisa_(quantidade, singular, plural) {
  return quantidade + ' ' + (quantidade === 1 ? singular : plural);
}

function formatarRazaoPesquisa_(quantidade, base) {
  if (!base) return 'Sem base';
  return quantidade + ' de ' + base + ' | ' + Math.round((quantidade / base) * 100) + '%';
}

function corTermometroPesquisa_(quantidade, base) {
  const c = CONFIG_PESQUISA_SATISFACAO.CORES;
  if (!base || quantidade === 0) return c.VERDE;
  return quantidade / base >= 0.25 ? c.VERMELHO : c.LARANJA;
}

function prepararAbaPesquisa_(aba) {
  if (aba.getMaxColumns() < CONFIG_PESQUISA_SATISFACAO.TOTAL_COLUNAS) {
    aba.insertColumnsAfter(
      aba.getMaxColumns(),
      CONFIG_PESQUISA_SATISFACAO.TOTAL_COLUNAS - aba.getMaxColumns()
    );
  }

  aba.setFrozenRows(0);
  aba.setFrozenColumns(0);

  const filtro = aba.getFilter();
  if (filtro) {
    filtro.remove();
  }

  aba.getCharts().forEach(function(grafico) {
    aba.removeChart(grafico);
  });

  aba.getRange(1, 1, aba.getMaxRows(), aba.getMaxColumns()).breakApart();
  aba.clear();
  aba.clearConditionalFormatRules();
  aba.showSheet();
  aba.setHiddenGridlines(true);
  aba.setTabColor(CONFIG_PESQUISA_SATISFACAO.CORES.TURQUESA);
}

  aba.hideColumns(27);

      .build(),
    SpreadsheetApp.newConditionalFormatRule()
      .whenTextContains('Durante')

    SpreadsheetApp.newConditionalFormatRule()
      .whenTextContains('Pós')
      .setBackground(c.VERDE_CLARO)
      .setRanges([faixaMomento])
      .build(),
    SpreadsheetApp.newConditionalFormatRule()
      .whenTextEqualTo('Sim')
      .setBackground(c.VERDE_CLARO)
      .setRanges([faixaClareza])
      .build(),
    SpreadsheetApp.newConditionalFormatRule()
      .whenTextContains('Parcial')
      .setBackground(c.AMARELO_CLARO)
      .setRanges([faixaClareza])
      .build(),
    SpreadsheetApp.newConditionalFormatRule()
      .whenTextEqualTo('Não')

      .build()

function obterOuCriarAbaAnalitica_(planilha, nome) {
  return planilha.getSheetByName(nome) || planilha.insertSheet(nome);
}

function prepararAbaAnalitica_(aba, totalColunas) {
  if (aba.getMaxColumns() < totalColunas) {
    aba.insertColumnsAfter(aba.getMaxColumns(), totalColunas - aba.getMaxColumns());
  }
  aba.setFrozenRows(0);
  aba.setFrozenColumns(0);
  if (aba.getFilter()) aba.getFilter().remove();
  aba.getCharts().forEach(function(grafico) { aba.removeChart(grafico); });
  aba.getRange(1, 1, aba.getMaxRows(), aba.getMaxColumns()).breakApart();
  aba.clear();
  aba.clearConditionalFormatRules();
  aba.setHiddenGridlines(true);
}

function aplicarRotulosEmNegritoPesquisa_(celula, texto, corTexto) {
  const construtor = SpreadsheetApp.newRichTextValue().setText(texto);
  const estiloNormal = SpreadsheetApp.newTextStyle().setForegroundColor(corTexto).build();
  const estiloNegrito = SpreadsheetApp.newTextStyle().setForegroundColor(corTexto).setBold(true).build();
  construtor.setTextStyle(0, texto.length, estiloNormal);
  const expressao = /(^|\n)([^\n:]{2,45}:)/g;
  let resultado;
  while ((resultado = expressao.exec(texto)) !== null) {
    const inicio = resultado.index + (resultado[1] ? resultado[1].length : 0);
    construtor.setTextStyle(inicio, inicio + resultado[2].length, estiloNegrito);
  }
  celula.setRichTextValue(construtor.build());
}

function formatarComparacaoCurtaPesquisa_(atual, anterior, percentual) {
  if (atual === null || atual === undefined || !Number.isFinite(atual)) return 'sem resultado atual';
  if (anterior === null || anterior === undefined || !Number.isFinite(anterior)) return 'sem base anterior';
  const diferenca = Math.round((atual - anterior) * 10) / 10;
  if (diferenca === 0) return 'estável';
  const sinal = diferenca > 0 ? '+' : '';
  const unidade = percentual
    ? ' p.p.'
    : (Math.abs(diferenca) === 1 ? ' ponto' : ' pontos');
  return sinal + diferenca.toFixed(1).replace('.', ',') + unidade;
}

function formatarPercentualPesquisa_(valor) {
  return valor === null || valor === undefined || !Number.isFinite(valor)
    ? 'Sem base'
    : Math.round(valor) + '%';
}

function normalizarMomentoPesquisa_(valor) {
  const texto = normalizarPesquisa_(valor);
  if (texto.indexOf('pre obra') !== -1) return 'pre obra';
  if (texto.indexOf('durante') !== -1 && texto.indexOf('pos obra') !== -1) return 'durante ou pos obra';
  if (texto.indexOf('pos obra') !== -1) return 'pos obra';
  if (texto.indexOf('durante') !== -1) return 'durante a obra';
  return texto;
}

function escreverComposicaoMensalPesquisa_(aba, registros, rotuloPeriodo) {
  const c = CONFIG_PESQUISA_SATISFACAO.CORES;
  const origens = { 'Histórico 1.0': 0, 'Histórico 2.0': 0, 'Atual 3.0': 0 };
  const modalidades = { 'Pesquisa direta': 0, 'Vinculada a vistoria cautelar': 0 };
  registros.forEach(function(registro) {
    origens[classificarOrigemPesquisa_(registro)]++;
    modalidades[classificarModalidadePesquisa_(registro)]++;
  });

  aba.getRange('I41:P41').merge().setValue('COMPOSIÇÃO DO MÊS')
    .setBackground(c.ROXO).setFontColor(c.BRANCO).setFontWeight('bold').setFontSize(10);

  aba.getRange('M42:P42').merge().setValue('Modalidade da pesquisa')
    .setBackground(c.AZUL_CLARO).setFontColor(c.TEXTO).setFontWeight('bold').setHorizontalAlignment('center');
  const linhasOrigem = Object.keys(origens).map(function(chave) { return [chave, origens[chave]]; });
  const linhasModalidade = Object.keys(modalidades).map(function(chave) { return [chave, modalidades[chave]]; });

    aba.getRange('L' + linha).setValue(item[1]).setNumberFormat('0').setHorizontalAlignment('center').setFontWeight('bold');

function classificarOrigemPesquisa_(registro) {
  const id = normalizarPesquisa_(registro.id);
  const origem = normalizarPesquisa_(registro.origemRegistro);
  if (id.indexOf('hist 1 0') === 0 || origem.indexOf('1 0') !== -1) return 'Histórico 1.0';
  if (id.indexOf('hist 2 0') === 0 || origem.indexOf('2 0') !== -1) return 'Histórico 2.0';
  return 'Atual 3.0';
}

function classificarModalidadePesquisa_(registro) {
  return normalizarPesquisa_(registro.procedimento).indexOf('cautelar') !== -1
    ? 'Vinculada a vistoria cautelar'
    : 'Pesquisa direta';
}

function agruparRegistrosPorMes_(registros, fusoHorario) {
  const mapa = {};
  registros.forEach(function(registro) {
    if (!(registro.data instanceof Date) || isNaN(registro.data.getTime())) return;
    const chave = Utilities.formatDate(registro.data, fusoHorario || 'America/Sao_Paulo', 'yyyy-MM');
    if (!mapa[chave]) mapa[chave] = [];
    mapa[chave].push(registro);
  });
  return Object.keys(mapa).sort().map(function(chave) {
    const partes = chave.split('-');
    const meses = ['janeiro', 'fevereiro', 'março', 'abril', 'maio', 'junho', 'julho', 'agosto', 'setembro', 'outubro', 'novembro', 'dezembro'];
    return {
      chave: chave,
      rotulo: meses[Number(partes[1]) - 1] + ' de ' + partes[0],
      registros: mapa[chave]
    };
  });
}

function obterPeriodoGravadoNaAbaPesquisa_(aba) {
  let valor = '';
  try {
    valor = textoPesquisa_(aba.getRange(CONFIG_PESQUISA_SATISFACAO.CELULA_PERIODO_BI).getDisplayValue());
  } catch (erro) {
    valor = '';
  }
  if (valor && normalizarPesquisa_(valor).indexOf('termometro dos procedimentos') === -1) {
    return valor;
  }
  return textoPesquisa_(PropertiesService.getDocumentProperties().getProperty(
    CONFIG_PESQUISA_SATISFACAO.PROPRIEDADE_PERIODO
  ));
}

function construirContextoMensalPesquisa_(registros, fusoHorario, periodoSolicitado) {
  const periodos = agruparRegistrosPorMes_(registros, fusoHorario);
  if (!periodos.length) {
    return {
      periodos: [],
      atual: { chave: '', rotulo: 'Sem período', registros: [] },
      anterior: null
    };
  }
  const solicitadoNormalizado = normalizarPesquisa_(periodoSolicitado);
  let indiceAtual = periodos.length - 1;
  if (solicitadoNormalizado) {
    const encontrado = periodos.findIndex(function(periodo) {
      return normalizarPesquisa_(periodo.chave) === solicitadoNormalizado
        || normalizarPesquisa_(periodo.rotulo) === solicitadoNormalizado;
    });
    if (encontrado !== -1) indiceAtual = encontrado;
  }
  return {
    periodos: periodos,
    atual: periodos[indiceAtual],
    anterior: indiceAtual > 0 ? periodos[indiceAtual - 1] : null
  };
}

function contarClassificacoesPesquisa_(registros) {
  const resultado = { Elogio: 0, Sugestão: 0, Solicitação: 0, Reclamação: 0, Neutra: 0 };
  registros.forEach(function(registro) {
    const classe = resumirClassificacaoPesquisa_(registro.classificacao);
    if (resultado[classe] !== undefined) resultado[classe]++;
  });
  return resultado;
}

function resumirClassificacaoPesquisa_(classificacao) {
  const texto = normalizarPesquisa_(classificacao);
  if (texto.indexOf('elogio') === 0) return 'Elogio';
  if (texto.indexOf('sugestao') === 0) return 'Sugestão';
  if (texto.indexOf('solicitacao') === 0) return 'Solicitação';
  if (texto.indexOf('reclamacao') === 0) return 'Reclamação';
  if (texto.indexOf('neutra') === 0) return 'Neutra';
  return classificacao ? 'Outra classificação' : '';
}

function contarTemasPesquisa_(registros) {
  const resultado = {};
  obterRegrasTemasPesquisa_().forEach(function(regra) { resultado[regra[0]] = 0; });
  registros.forEach(function(registro) {
    const temas = registro.temas ? registro.temas.split(', ') : [];
    temas.forEach(function(tema) {
      if (resultado[tema] !== undefined) resultado[tema]++;
    });
  });
  return resultado;
}

function buscarValorPesquisa_(linha, mapa, titulos) {
  for (let i = 0; i < titulos.length; i++) {
    const indice = mapa[normalizarPesquisa_(titulos[i])];
    if (indice !== undefined) {
      const valor = linha[indice];
      if (valor !== '' && valor !== null && valor !== undefined) {
        return valor;
      }
    }
  }
  return null;
}

function normalizarPesquisa_(valor) {
  return String(valor || '')
    .normalize('NFD')
    .replace(/[\u0300-\u036f]/g, '')
    .toLowerCase()
    .replace(/[^a-z0-9]+/g, ' ')
    .trim();
}

function textoPesquisa_(valor) {
  if (valor === null || valor === undefined || String(valor).trim() === '') {
    return '';
  }
  return String(valor).trim();
}

function formatarTelefonePesquisa_(valor) {
  if (valor === null || valor === undefined || String(valor).trim() === '') {
    return '';
  }
  if (typeof valor === 'number') {
    return String(Math.trunc(valor));
  }
  return String(valor).trim();
}

function numeroPesquisa_(valor) {
  if (valor === null || valor === undefined || valor === '') {
    return null;
  }
  const numero = typeof valor === 'number'
    ? valor
    : Number(String(valor).replace(',', '.'));
  return Number.isFinite(numero) ? numero : null;
}

function converterDataPesquisa_(valor) {
  if (valor instanceof Date && !isNaN(valor.getTime())) {
    return valor;
  }
  if (typeof valor === 'number' && Number.isFinite(valor)) {
    return new Date(Math.round((valor - 25569) * 86400 * 1000));
  }
  if (typeof valor === 'string' && valor.trim()) {
    const partes = valor.trim().match(/^(\d{1,2})[\/\-](\d{1,2})[\/\-](\d{4})/);
    if (partes) {
      return new Date(Number(partes[3]), Number(partes[2]) - 1, Number(partes[1]));
    }
    const tentativa = new Date(valor);
    if (!isNaN(tentativa.getTime())) {
      return tentativa;
    }
  }
  return null;
}

function numerosValidosPesquisa_(valores) {
  return valores.filter(function(valor) {
    return typeof valor === 'number' && Number.isFinite(valor);
  });
}

function mediaPesquisa_(valores) {
  if (!valores.length) {
    return null;
  }
  return valores.reduce(function(total, valor) {
    return total + valor;
  }, 0) / valores.length;
}

function formatarMediaPesquisa_(valor) {
  if (valor === null) {
    return 'Sem dados';
  }
  return valor.toFixed(1).replace('.', ',') + ' / 10';
}
